function sendFeedback() { try { const parentWindow = window.parent; if (parentWindow && parentWindow !== window && parentWindow.document) { const launcher = parentWindow.document.getElementById("contact-launcher"); const dialog = parentWindow.document.getElementById("contact-dialog"); const category = parentWindow.document.getElementById("feedback-category"); if (launcher) { launcher.click(); } else if (dialog) { dialog.setAttribute("aria-hidden", "false"); } if (category) { category.focus(); } setStatus("Opening the VAVA contact form..."); return; } } catch (error) { console.warn("Unable to open the parent contact form:", error); } setStatus("The VAVA contact form is unavailable in this context. Open the main app window to send feedback.", true); }
